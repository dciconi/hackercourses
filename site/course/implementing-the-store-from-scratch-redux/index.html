    In Dan Abramov's excellent egghead redux course [0] he implements the `createStore` from scratch which is the core of redux, it's simple enough to post here:

    const createStore = (reducer) => { let state; let listeners = [];

    const getState = () => state;

    const dispatch = (action) => { state = reducer(state, action); listeners.map(listener => listener()); };

    const subscribe = (listener) => { listeners.push(listener); // unsubscribe return () => { listeners = listeners.filter(l => l !== listener) }; };

    // populate initial state dispatch({});

    return { getState, dispatch, subscribe }; };
